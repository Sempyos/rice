        initial: "", 
        cmd: "date +%M",

        initial: "", 
        cmd: "date +%m",

fn imageButton( cmd, class, path, fill, scale) {
    eventbox(#{ onclick: cmd, class: "btn23"}, [
    image(#{ class: class, path: path, fill_svg: fill, image_width: scale, can_shrink: true,  content_fit: "scaledown"}),
    ]);
};
